# A step whose trace is forgotten before it is checked runs, and one already checked stays as it was.

step compile in=main.c out=main.o
step link in=main.o out=app
trace compile in=main.c:c1 out=main.o:o1
trace link in=main.o:o1 out=app:a1

request link
need main.c for compile -> c1
need main.o for compile -> o1
forget compile
forget link
run link no_trace -> ok
need app for link -> a1
built link in=main.o:o1 out=app:a1
done
