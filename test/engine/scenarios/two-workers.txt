# Events handed out stay out until answered: a second worker gets the next step that is ready, then idles while everything left waits on the first.

step compile_a in=a.c out=a.o
step compile_b in=b.c out=b.o
step link in=a.o,b.o out=app

request link
need a.c for compile_a
need b.c for compile_b
idle
give b.c:b1
run compile_b no_trace
idle
give a.c:a1
run compile_a no_trace
finish compile_a ok
need a.o for compile_a -> ao1
built compile_a in=a.c:a1 out=a.o:ao1
idle
finish compile_b ok
need b.o for compile_b -> bo1
built compile_b in=b.c:b1 out=b.o:bo1
run link no_trace -> ok
need app for link -> app1
built link in=a.o:ao1,b.o:bo1 out=app:app1
done
